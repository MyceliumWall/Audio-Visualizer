#!/usr/bin/env python3
"""
Install:  pip install numpy scipy sounddevice soundfile
Run:      python audio_visualizer.py
"""
import math
import threading
import time
import tkinter as tk
import tkinter.font as tkfont
from tkinter import filedialog

import numpy as np
import sounddevice as sd
import soundfile as sf
from scipy.signal import butter, sosfilt

# ---------- look (same palette as the HTML version) ----------
BG, INK, MUTE = "#000000", "#c46a6a", "#8f3a3a"
PANEL, LINE, WAVE, ONINK = "#0a0202", "#6B0505", "#780606", "#ffd6d6"
GLOW_OUTER, GLOW_INNER = "#2a0202", "#4d0404"

SENS = 0.2 / 0.6  # wave gain, fixed at the lowest sensitivity
FFT = 2048        # samples analysed per frame (same as analyser fftSize)
WARMUP = 6144     # extra samples fed to the filters so their start-up ringing is discarded
PTS = 300         # points per drawn wave

# ---------- Rainmeter-style visualizer modes ----------
MODES = ["Waves", "Bars", "Mirror bars", "Mountain", "Ring"]
NBARS = 40                  # frequency bars used by every non-wave mode
F_LO, F_HI = 40, 10000      # spectrum range (Hz), log-spaced across the bars

# three frequency ranges, drawn top to bottom.
# win = how many samples of the signal each wave shows (shorter for higher ranges)
BANDS = [
    {"name": "High", "range": "2\u201310 kHz",      "lo": 2000, "hi": 10000, "win": 256},
    {"name": "Mid",  "range": "250 Hz\u20132 kHz",  "lo": 250,  "hi": 2000,  "win": 640},
    {"name": "Low",  "range": "40\u2013250 Hz",     "lo": 40,   "hi": 250,   "win": 1536},
]
for b in BANDS:
    b.update(show=True, lvl=0.05, agc=1.0, sos=None, buf=np.zeros(FFT))


class Player:
    """Loops an audio file through the speakers and keeps a ring of the latest mono samples."""

    def __init__(self):
        self.stream = None
        self.data = None
        self.sr = 44100
        self.pos = 0
        self.volume = 0.8
        self.lock = threading.Lock()
        self.ring = np.zeros(FFT + WARMUP, dtype=np.float32)

    @property
    def playing(self):
        return self.stream is not None

    def _callback(self, outdata, frames, _time, _status):
        n = len(self.data)
        idx = (self.pos + np.arange(frames)) % n
        chunk = self.data[idx]
        self.pos = (self.pos + frames) % n
        # visuals read the signal before the volume control, like the HTML version
        mono = chunk.mean(axis=1)
        with self.lock:
            self.ring = np.concatenate((self.ring[frames:], mono))[-len(self.ring):]
        outdata[:] = chunk * self.volume

    def load_and_play(self, path):
        self.stop()
        data, sr = sf.read(path, dtype="float32", always_2d=True)
        if data.shape[1] > 2:
            data = data[:, :2]
        self.data, self.sr, self.pos = data, sr, 0
        self.ring[:] = 0
        for b in BANDS:
            hi = min(b["hi"], 0.45 * sr)
            nyq = sr / 2
            # band-pass = 4th-order high-pass + 4th-order low-pass (24 dB/octave edges)
            b["sos"] = np.vstack([
                butter(4, b["lo"] / nyq, "highpass", output="sos"),
                butter(4, hi / nyq, "lowpass", output="sos"),
            ])
            b["lvl"], b["agc"] = 0.05, 1.0
        self.stream = sd.OutputStream(samplerate=sr, channels=data.shape[1],
                                      callback=self._callback)
        self.stream.start()

    def stop(self):
        if self.stream is not None:
            try:
                self.stream.stop()
                self.stream.close()
            except Exception:
                pass
            self.stream = None

    def snapshot(self):
        with self.lock:
            return self.ring.copy()


class App:
    def __init__(self, root):
        self.root = root
        root.title("Audio visualizer")
        root.configure(bg=BG)
        root.geometry("1000x620")
        root.minsize(560, 360)

        fams = set(tkfont.families())
        self.family = next((f for f in ("Share Tech Mono", "Menlo", "Consolas", "Courier New")
                            if f in fams), "Courier")
        self.font = (self.family, 10)
        self.font_bold = (self.family, 10, "bold")

        self.player = Player()
        self.W = self.H = 1
        self.start_time = time.time()

        # spectrum state shared by the Rainmeter-style modes
        self.mode = tk.StringVar(value=MODES[0])
        self.bars = np.zeros(NBARS)
        self.peak = 1e-3
        self.window = np.hanning(FFT)

        # ---- canvas ----
        self.cv = tk.Canvas(root, bg=BG, highlightthickness=0)
        self.cv.pack(fill="both", expand=True)
        self.cv.bind("<Configure>", self._resize)

        # ---- control bar ----
        bar = tk.Frame(root, bg=PANEL, highlightbackground=LINE, highlightthickness=1)
        bar.pack(fill="x", side="bottom")
        pad = dict(padx=6, pady=10)

        self.btn(bar, "Open audio file", self.open_file).pack(side="left", **pad)

        # Rainmeter menu: picks the visualizer style (sits right next to the open button)
        rm_btn, rm_menu = self.menubtn(bar, "Rainmeter \u25be")
        for name in MODES:
            rm_menu.add_radiobutton(label=name, variable=self.mode, value=name,
                                    command=self._mode_changed)
        rm_btn.pack(side="left", **pad)

        # Waves menu: toggles the three wave ranges (only used in Waves mode)
        self.show_vars = []
        self.waves_btn, menu = self.menubtn(bar, "Waves \u25be")
        for b in BANDS:
            var = tk.BooleanVar(value=True)
            menu.add_checkbutton(label=f"{b['name']} \u00b7 {b['range']}", variable=var,
                                 command=lambda b=b, v=var: b.update(show=v.get()))
            self.show_vars.append(var)
        self.waves_btn.pack(side="left", **pad)

        self.btn(bar, "Stop", self.stop).pack(side="left", **pad)

        tk.Label(bar, text="Volume", font=self.font, fg=MUTE, bg=PANEL).pack(side="left", padx=(24, 4))
        self.vol = tk.Scale(bar, from_=0, to=1, resolution=0.01, orient="horizontal", length=120,
                            showvalue=False, bg=PANEL, fg=INK, troughcolor=LINE,
                            highlightthickness=0, bd=0, activebackground=WAVE,
                            command=self._set_volume)
        self.vol.set(0.8)
        self.vol.pack(side="left", pady=6)

        self.status = tk.Label(bar, text="", font=self.font, fg=MUTE, bg=PANEL, anchor="e")
        self.status.pack(side="right", padx=12)

        root.protocol("WM_DELETE_WINDOW", self.close)
        self.frame()

    # ---------- ui helpers ----------
    def btn(self, parent, text, cmd):
        return tk.Button(parent, text=text, command=cmd, font=self.font_bold, fg=INK, bg=PANEL,
                         activebackground=WAVE, activeforeground=ONINK, relief="solid", bd=1,
                         padx=10, pady=4, cursor="hand2")

    def menubtn(self, parent, text):
        """A dropdown button + its menu, styled like the other controls."""
        mb = tk.Menubutton(parent, text=text, font=self.font_bold, fg=INK, bg=PANEL,
                           activebackground=LINE, activeforeground=ONINK,
                           disabledforeground=MUTE, relief="solid", bd=1, padx=10, pady=4)
        menu = tk.Menu(mb, tearoff=0, bg=PANEL, fg=INK, activebackground=WAVE,
                       activeforeground=ONINK, selectcolor=ONINK, font=self.font)
        mb["menu"] = menu
        return mb, menu

    def say(self, msg=""):
        self.status.config(text=msg)

    def _resize(self, e):
        self.W, self.H = max(1, e.width), max(1, e.height)

    def _set_volume(self, v):
        self.player.volume = float(v)

    def _mode_changed(self):
        # the wave toggles only apply to the Waves style
        self.waves_btn.config(state="normal" if self.mode.get() == "Waves" else "disabled")

    # ---------- actions ----------
    def open_file(self):
        path = filedialog.askopenfilename(
            title="Open audio file",
            filetypes=[("Audio", "*.wav *.flac *.ogg *.oga *.mp3 *.aiff *.aif"), ("All files", "*.*")])
        if not path:
            return
        try:
            self.player.load_and_play(path)
            self.bars[:] = 0
            self.peak = 1e-3
            self.say("Playing " + path.replace("\\", "/").split("/")[-1])
        except Exception:
            self.player.stop()
            self.say("This file could not be played.")

    def stop(self):
        self.player.stop()
        self.say("")

    def close(self):
        self.player.stop()
        self.root.destroy()

    # ---------- analysis ----------
    def analyse(self, snap):
        """Band-pass the latest samples and update each band's level, gain and start offset."""
        for b in BANDS:
            y = sosfilt(b["sos"], snap)[-FFT:]
            b["buf"] = y
            rms = float(np.sqrt(np.mean(y * y)))
            b["lvl"] += (rms - b["lvl"]) * (0.25 if rms > b["lvl"] else 0.015)  # fast attack, slow release
            b["agc"] = min(40.0, max(0.4, 0.25 / max(b["lvl"], 0.003)))          # each range levels itself
            # start each wave at a rising zero-crossing so it holds steady on screen
            lim = FFT - b["win"]
            cross = np.nonzero((y[:-1] < 0) & (y[1:] >= 0))[0] + 1
            cross = cross[cross < lim]
            b["start"] = int(cross[0]) if len(cross) else 0

    def spectrum(self, snap):
        """FFT the latest samples into NBARS log-spaced bars with fast attack, slow release."""
        sr = self.player.sr
        mag = np.abs(np.fft.rfft(snap[-FFT:] * self.window)) / (FFT / 4)
        freqs = np.fft.rfftfreq(FFT, 1 / sr)
        edges = np.geomspace(F_LO, min(F_HI, 0.45 * sr), NBARS + 1)
        centers = np.sqrt(edges[:-1] * edges[1:])
        vals = np.interp(centers, freqs, mag)           # fallback for bars that span no FFT bin
        for i in range(NBARS):
            m = (freqs >= edges[i]) & (freqs < edges[i + 1])
            if m.any():
                vals[i] = mag[m].max()
        vals *= np.sqrt(centers / 200)                  # tilt: music falls off towards the highs
        self.peak = max(self.peak * 0.995, float(vals.max()), 1e-4)  # slow auto-gain
        v = np.sqrt(vals / self.peak)
        self.bars += (v - self.bars) * np.where(v > self.bars, 0.6, 0.12)

    # ---------- drawing ----------
    def frame(self):
        self.draw()
        self.root.after(16, self.frame)

    def draw(self):
        cv, W, H = self.cv, self.W, self.H
        active = self.player.playing
        mode = self.mode.get()
        cv.delete("all")
        cy = H / 2

        # shared grid
        for k in range(1, 8):
            y = H * k / 8
            cv.create_line(0, y, W, y, fill=LINE)
        for k in range(1, 12):
            x = W * k / 12
            cv.create_line(x, 0, x, H, fill=LINE)
        cv.create_line(0, cy, W, cy, fill=LINE)

        now = (time.time() - self.start_time) * 1000
        if active:
            if mode == "Waves":
                self.analyse(self.player.snapshot())
            else:
                self.spectrum(self.player.snapshot())
        else:
            cv.create_text(W / 2, H / 2, text="Pick a sound source below to see it move.",
                           fill=MUTE, font=(self.family, 16))
            # gentle idle motion for the spectrum styles
            self.bars = 0.08 + 0.05 * np.sin(now / 700 + np.arange(NBARS) * 0.35)

        if mode == "Waves":
            self.draw_waves(active, now)
        else:
            self.legend(0, mode, f"{F_LO} Hz\u2013{F_HI // 1000} kHz", max(0.8, min(W, H) * 0.0022))
            {"Bars": self.draw_bars, "Mirror bars": self.draw_mirror,
             "Mountain": self.draw_mountain, "Ring": self.draw_ring}[mode]()

    def legend(self, row, name, rng, lw):
        ly = 12 + row * 18
        self.cv.create_line(14, ly + 7, 38, ly + 7, fill=WAVE, width=lw, capstyle="round")
        self.cv.create_text(48, ly, text=f"{name} \u00b7 {rng}", fill=INK,
                            anchor="nw", font=(self.family, 9))

    # ----- original three-range waves -----
    def draw_waves(self, active, now):
        cv, W, H = self.cv, self.W, self.H
        cy = H / 2
        m = min(W, H)
        row = 0
        for li, b in enumerate(BANDS):
            if not b["show"]:
                continue
            # same colour, so thickness tells the ranges apart (each range below is 10% thicker)
            lw = max(0.8, m * 0.0022) * [1.1, 1.21, 1.331][li]

            self.legend(row, b["name"], b["range"], lw)
            row += 1

            # wave points
            p = np.arange(PTS + 1)
            if active:
                src = b["buf"][b["start"] + (p / PTS * (b["win"] - 1)).astype(int)]
                v = np.tanh(src * b["agc"] * SENS * 2)
            else:
                v = np.sin(now / 700 + p * (0.05 + li * 0.01) + li * 0.8) * 0.06
            xs = p / PTS * W
            ys = cy + v * H * 0.34
            coords = np.column_stack((xs, ys)).ravel().tolist()

            # soft red glow, red body, then a thin white core on top
            kw = dict(capstyle="round", joinstyle="round")
            cv.create_line(coords, fill=GLOW_OUTER, width=lw * 6, **kw)
            cv.create_line(coords, fill=GLOW_INNER, width=lw * 3, **kw)
            cv.create_line(coords, fill=WAVE, width=lw, **kw)
            cv.create_line(coords, fill="#ffffff", width=max(0.6, lw * 0.4), **kw)

    # ----- Rainmeter-style spectrum modes (same glow / body / white-core layering) -----
    def _bar(self, x0, y0, x1, y1, g):
        """One bar: soft glow layers, red body, thin white core."""
        cv = self.cv
        cv.create_rectangle(x0 - g * 2, y0 - g * 2, x1 + g * 2, y1 + g * 2,
                            fill=GLOW_OUTER, outline="")
        cv.create_rectangle(x0 - g, y0 - g, x1 + g, y1 + g, fill=GLOW_INNER, outline="")
        cv.create_rectangle(x0, y0, x1, y1, fill=WAVE, outline="")
        cx = (x0 + x1) / 2
        cv.create_line(cx, y0, cx, y1, fill="#ffffff", width=max(0.6, (x1 - x0) * 0.12))

    def _bar_geometry(self):
        slot = self.W / NBARS
        bw = slot * 0.6
        g = max(1.0, (slot - bw) * 0.17)
        return slot, bw, g

    def draw_bars(self):
        H = self.H
        slot, bw, g = self._bar_geometry()
        base, maxh = H * 0.94, H * 0.8
        for i, v in enumerate(self.bars):
            x0 = i * slot + (slot - bw) / 2
            self._bar(x0, base - max(2.0, v * maxh), x0 + bw, base, g)

    def draw_mirror(self):
        H = self.H
        cy = H / 2
        slot, bw, g = self._bar_geometry()
        for i, v in enumerate(self.bars):
            x0 = i * slot + (slot - bw) / 2
            h = max(1.5, v * H * 0.38)
            self._bar(x0, cy - h, x0 + bw, cy + h, g)

    def draw_mountain(self):
        cv, W, H = self.cv, self.W, self.H
        lw = max(0.8, min(W, H) * 0.0022) * 1.21
        base, maxh = H * 0.94, H * 0.8
        n = 200
        v = np.interp(np.linspace(0, NBARS - 1, n), np.arange(NBARS), self.bars)
        v = np.convolve(np.pad(v, (4, 4), mode="edge"), np.ones(9) / 9, mode="valid")
        xs = np.linspace(0, W, n)
        ys = base - np.maximum(v, 0.01) * maxh
        pts = np.column_stack((xs, ys)).ravel().tolist()
        cv.create_polygon(pts + [W, base, 0, base], fill=GLOW_OUTER, outline="")
        kw = dict(capstyle="round", joinstyle="round")
        cv.create_line(pts, fill=GLOW_INNER, width=lw * 3, **kw)
        cv.create_line(pts, fill=WAVE, width=lw, **kw)
        cv.create_line(pts, fill="#ffffff", width=max(0.6, lw * 0.4), **kw)

    def draw_ring(self):
        cv, W, H = self.cv, self.W, self.H
        cx, cy = W / 2, H / 2
        m = min(W, H)
        r0, rmax = m * 0.16, m * 0.30
        lw = max(1.5, m * 0.006)
        cv.create_oval(cx - r0, cy - r0, cx + r0, cy + r0, outline=LINE)
        total = NBARS * 2  # mirrored left/right, lows at the top
        kw = dict(capstyle="round")
        for k in range(total):
            i = k if k < NBARS else total - 1 - k
            ang = -math.pi / 2 + 2 * math.pi * k / total
            r1 = r0 + 3 + self.bars[i] * rmax
            c, s = math.cos(ang), math.sin(ang)
            x0, y0, x1, y1 = cx + c * r0, cy + s * r0, cx + c * r1, cy + s * r1
            cv.create_line(x0, y0, x1, y1, fill=GLOW_INNER, width=lw * 2, **kw)
            cv.create_line(x0, y0, x1, y1, fill=WAVE, width=lw, **kw)
            cv.create_line(x0, y0, x1, y1, fill="#ffffff", width=max(0.6, lw * 0.35), **kw)


def main():
    root = tk.Tk()
    App(root)
    root.mainloop()


if __name__ == "__main__":
    main()
